cmake_minimum_required(VERSION 3.16)
project(finite_step_two_grid LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED)
add_library(tgi_headers INTERFACE)
target_include_directories(tgi_headers INTERFACE src)
target_link_libraries(tgi_headers INTERFACE OpenMP::OpenMP_CXX)
target_compile_definitions(tgi_headers INTERFACE TGI_RESULTS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/results\")
target_compile_options(tgi_headers INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion -Werror)
function(tgi_add_program target file)
    add_executable(${target} ${file})
    target_link_libraries(${target} PRIVATE tgi_headers)
endfunction()
tgi_add_program(experiment1_finite_path_comparison experiments/experiment1_finite_path_comparison.cpp)
tgi_add_program(experiment2_spectral_path experiments/experiment2_spectral_path.cpp)
tgi_add_program(experiment4_local_diagnostic_export experiments/experiment4_local_diagnostic_export.cpp)
tgi_add_program(experiment5_robustness experiments/experiment5_robustness.cpp)
tgi_add_program(experiment6_endpoint_comparison experiments/experiment6_endpoint_comparison.cpp)
